def fillna_unknown_dtype_col(X):
    if X.dtype == int or X.dtype == np.int64:
        X = X.fillna(value=-1)
    elif X.dtype == float:
        X = X.fillna(value=-1)
    elif X.dtype == bool:
        X = X.fillna(value=False)
    elif X.dtype == object:
        X = X.fillna(value='missing')
    else:
        print(X.dtype)
    return X
for feat in X.columns:
